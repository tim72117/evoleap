# Dockerfile：照著做，就做出一模一樣的便當
FROM node:22-slim          ← 底下墊什麼（作業系統＋Node）
WORKDIR /app
COPY package*.json ./      ← 先放套件清單
RUN npm ci                 ← 裝好套件
COPY . .                   ← 放進你的程式
CMD ["node", "server.js"]  ← 打開時要做什麼
你的程式（server.js、網頁）
套件（npm ci 裝好的）
Node 22
精簡的作業系統
